Looking at the screenshot, the problems are concrete:

**The jet is unreadable because of the camera angle, not the model.** The camera sits almost level behind the ship (`y=3.4`, look-ahead `-18`), so you see the flat rear face of the 2.4-wide fuselage and nothing else — the wings, tails and canopy are all hidden behind that cross-section. The big white circle is the *engine glow sphere*: unshaded, additive, `radius 1.0`, sitting centred at `z=6.8` directly in front of the camera. It's the biggest, brightest object on screen.

**The canyon reads as a pale city skyline,** not rock: uniform `ROCK_COLOR` on every block, vertical monoliths 1:1 aspect with no strata, no colour variation, and the corridor is 180 units wide so the walls never loom.

**The whole frame is washed out:** `fog_density = 0.004` plus `ambient_light_energy = 0.7` from the sky flattens all shading to the same pale gray-blue, and there's no tonemapping, no shadows, no contrast adjustment.

Fixes: chase cam raised into a shallow top-down 3/4 so the airframe silhouette reads; twin small additive nozzles instead of one sphere; layered strata with tonal variation + triplanar noise; narrower, taller, shadow-casting corridor; ACES tonemapping and a CPU particle trail.

func _setup_canyon() -> void:
	var ground_length := CORRIDOR_LENGTH * 2.0
	_ground_mesh.size = Vector3(700.0, 14.0, ground_length)
	_static_box(
		Vector3(0.0, -7.0, -CORRIDOR_LENGTH * 0.5),
		_ground_mesh.size,
		_ground_mesh,
		_ground_material
	)

	for i in range(WALL_COUNT):
		var segment_z := -70.0 - float(i) * WALL_SPACING
		var side_shift := rng.randf_range(-14.0, 14.0)

		for side in [-1.0, 1.0]:
			var base_width := 40.0 + rng.randf_range(10.0, 30.0)
			var height_left := WALL_HEIGHT * rng.randf_range(0.7, 1.25)
			var y := -6.0

			for s in range(STRATA_COUNT):
				if y > height_left:
					break
				var slab_height := rng.randf_range(24.0, 46.0)
				var width := base_width - float(s) * 2.5 + rng.randf_range(-5.0, 5.0)
				var inset := float(s) * STRATA_TAPER + rng.randf_range(-4.0, 6.0)
				var x := side * (CORRIDOR_HALF_WIDTH + inset + width * 0.5 + side_shift)
				var z := segment_z + rng.randf_range(-9.0, 9.0)
				var material := _tone_materials[rng.randi_range(0, _tone_materials.size() - 1)]

				_static_box(
					Vector3(x, y + slab_height * 0.5, z),
					Vector3(width, slab_height, WALL_SPACING * rng.randf_range(0.7, 0.95)),
					_strata_mesh,
					material,
					Vector3(rng.randf_range(-0.05, 0.05), rng.randf_range(-0.07, 0.07), 0.0)
				)
				y += slab_height

			# talus boulders piled against the wall base
			for r in range(TALUS_ROCK_COUNT):
				_boulder(
					Vector3(
						side * (CORRIDOR_HALF_WIDTH + rng.randf_range(-2.0, 14.0)),
						rng.randf_range(-2.0, 3.0),
						segment_z + rng.randf_range(-30.0, 30.0)
					),
					rng.randf_range(2.5, 6.5)
				)

	# loose rocks scattered across the corridor floor
	for r in range(GROUND_ROCK_COUNT):
		_boulder(
			Vector3(
				rng.randf_range(-CORRIDOR_HALF_WIDTH, CORRIDOR_HALF_WIDTH),
				rng.randf_range(-1.0, 2.0),
				rng.randf_range(-CORRIDOR_LENGTH, 60.0)
			),
			rng.randf_range(1.6, 4.5)
		)

	# distant mesas beyond the canyon for depth
	for m in range(MESA_COUNT):
		var mesa_side := 1.0 if (m % 2 == 0) else -1.0
		var mesa_width := rng.randf_range(120.0, 260.0)
		var mesa_height := rng.randf_range(160.0, 320.0)
		var body := StaticBody3D.new()
		body.position = Vector3(
			mesa_side * rng.randf_range(280.0, 520.0),
			mesa_height * 0.5 - 20.0,
			-CORRIDOR_LENGTH * 0.5 + rng.randf_range(-200.0, 200.0)
		)
		body.rotation.y = rng.randf_range(-0.4, 0.4)

		var collision := CollisionShape3D.new()
		var shape := BoxShape3D.new()
		shape.size = Vector3(mesa_width, mesa_height, mesa_width)
		collision.shape = shape
		body.add_child(collision)

		var instance := MeshInstance3D.new()
		var mesh := BoxMesh.new()
		mesh.size = Vector3(mesa_width, mesa_height, mesa_width)
		instance.mesh = mesh
		instance.material_override = _mesa_material
		body.add_child(instance)

		add_child(body)


func _static_box(position: Vector3, size: Vector3, mesh: BoxMesh, material: Material, rotation: Vector3 = Vector3.ZERO) -> void:
	var body := StaticBody3D.new()
	body.position = position
	body.rotation = rotation

	# A Shape3D resource is not a Node: it must be wrapped in a CollisionShape3D
	# before it can be added to the body.
	var collision := CollisionShape3D.new()
	var shape := BoxShape3D.new()
	shape.size = size
	collision.shape = shape
	body.add_child(collision)

	var instance := MeshInstance3D.new()
	instance.mesh = mesh
	instance.material_override = material
	body.add_child(instance)

	add_child(body)


func _boulder(position: Vector3, size: float) -> void:
	var body := StaticBody3D.new()
	body.position = position
	body.scale = Vector3(size, size * rng.randf_range(0.6, 1.0), size)
	body.rotation = Vector3(0.0, rng.randf_range(0.0, TAU), 0.0)

	var collision := CollisionShape3D.new()
	var shape := SphereShape3D.new()
	shape.radius = 1.0
	collision.shape = shape
	body.add_child(collision)

	var instance := MeshInstance3D.new()
	instance.mesh = _boulder_mesh
	instance.material_override = _boulder_material
	body.add_child(instance)

	add_child(body)

func _place_camera(snap: bool, delta: float = 0.0) -> void:
	if player == null or camera == null:
		return

	var player_basis := player.global_transform.basis
	var target_position := player.global_position + player_basis * CAMERA_OFFSET
	var look_target := player.global_position + player_basis * CAMERA_LOOK_AHEAD

	if snap:
		camera.global_position = target_position
		camera.look_at(look_target, Vector3.UP)
		return

	var weight := 1.0 - exp(-CAMERA_SMOOTHING * delta)
	camera.global_position = camera.global_position.lerp(target_position, weight)
	camera.look_at(look_target, Vector3.UP)


func _update_fov(delta: float) -> void:
	if player == null or camera == null:
		return
	var target_fov := BASE_FOV
	if player.boosting:
		target_fov = BOOST_FOV
	camera.fov = lerpf(camera.fov, target_fov, 1.0 - exp(-4.0 * delta))
