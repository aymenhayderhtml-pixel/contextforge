func _setup_canyon() -> void:
	rng.randomize()

	_ground_box(
		Vector3(0.0, -6.0, -CORRIDOR_LENGTH * 0.5),
		Vector3(CORRIDOR_HALF_WIDTH * 3.0, 12.0, CORRIDOR_LENGTH * 2.0),
		GROUND_COLOR
	)

	for i in range(WALL_COUNT):
		var z := -60.0 - float(i) * WALL_SPACING
		var width := 30.0 + rng.randf_range(10.0, 34.0)
		var left_x := -(CORRIDOR_HALF_WIDTH + width * 0.5) + rng.randf_range(-WALL_JITTER, 0.0)
		var right_x := CORRIDOR_HALF_WIDTH + width * 0.5 + rng.randf_range(0.0, WALL_JITTER)
		var height := WALL_HEIGHT + rng.randf_range(-30.0, 45.0)

		_ground_box(
			Vector3(left_x, height * 0.5 - 8.0, z),
			Vector3(width, height, WALL_SPACING * 0.8),
			ROCK_COLOR
		)
		_ground_box(
			Vector3(right_x, height * 0.5 - 8.0, z),
			Vector3(width, height, WALL_SPACING * 0.8),
			ROCK_COLOR
		)


func _ground_box(position: Vector3, size: Vector3, color: Color) -> void:
	var body := StaticBody3D.new()
	body.position = position

	var shape := BoxShape3D.new()
	shape.size = size
	body.add_child(shape)

	var mesh_instance := MeshInstance3D.new()
	var mesh := BoxMesh.new()
	mesh.size = size
	mesh_instance.mesh = mesh
	mesh_instance.material_override = _flat_material(color)
	body.add_child(mesh_instance)

	add_child(body)


func _flat_material(color: Color) -> StandardMaterial3D:
	var material := StandardMaterial3D.new()
	material.albedo_color = color
	material.roughness = 1.0
	material.metallic = 0.0
	return material


func _setup_player() -> void:
	player = JET_SCRIPT.new()
	player.name = "PlayerJet"
	add_child(player)


func _place_camera(snap: bool, delta: float = 0.0) -> void:
	if player == null or camera == null:
		return

	var basis := player.global_transform.basis
	var target_position := player.global_position + basis * CAMERA_OFFSET
	var look_target := player.global_position + basis * CAMERA_LOOK_AHEAD

	if snap:
		camera.global_position = target_position
		camera.look_at(look_target, Vector3.UP)
		return

	var weight := 1.0 - exp(-CAMERA_SMOOTHING * delta)
	camera.global_position = camera.global_position.lerp(target_position, weight)
	camera.look_at(look_target, Vector3.UP)

const MAX_PITCH_RATE := 1.1
const MAX_YAW_RATE := 0.9
const MAX_ROLL_RATE := 2.2
const ANGULAR_SMOOTHING := 8.0
const FORWARD_SPEED := 95.0
const BOOST_MULTIPLIER := 1.7
const BOOST_ACCELERATION := 70.0
const BASE_ACCELERATION := 45.0
const MIN_ALTITUDE := 6.0
const MAX_ALTITUDE := 170.0
const MAX_SPEED := 220.0

const HULL_COLOR := Color(0.85, 0.88, 0.94)
const ACCENT_COLOR := Color(0.16, 0.32, 0.62)
const TRIM_COLOR := Color(0.20, 0.22, 0.27)
const GLASS_COLOR := Color(0.45, 0.85, 1.0, 0.65)

var speed := FORWARD_SPEED
var boosting := false

var _pitch := 0.0
var _yaw := 0.0
var _roll := 0.0
var _hull: Node3D
var _engine_glow: MeshInstance3D
var _engine_light: OmniLight3D
var _input_enabled := true


func _ready() -> void:
	collision_layer = 1
	collision_mask = 1
	_build_collision()
	_build_mesh()
	_build_engine_effect()


func _unhandled_input(event: InputEvent) -> void:
	var click := event as InputEventMouseButton
	if click != null and click.pressed:
		_input_enabled = not _input_enabled
		get_viewport().set_input_as_handled()


func _physics_process(delta: float) -> void:
	var pitch_input := 0.0
	var yaw_input := 0.0
	var roll_input := 0.0
	boosting = false

	if _input_enabled:
		pitch_input = Input.get_axis("pitch_down", "pitch_up")
		yaw_input = Input.get_axis("yaw_left", "yaw_right")
		roll_input = Input.get_axis("roll_left", "roll_right")
		boosting = Input.is_action_pressed("boost")

	_pitch = move_toward(_pitch, pitch_input * MAX_PITCH_RATE, MAX_PITCH_RATE * ANGULAR_SMOOTHING * delta)
	_yaw = move_toward(_yaw, yaw_input * MAX_YAW_RATE, MAX_YAW_RATE * ANGULAR_SMOOTHING * delta)
	_roll = move_toward(_roll, roll_input * MAX_ROLL_RATE, MAX_ROLL_RATE * ANGULAR_SMOOTHING * delta)

	rotate_object_local(Vector3.RIGHT, -_pitch * delta)
	rotate_object_local(Vector3.UP, -_yaw * delta)
	rotate_object_local(Vector3.FORWARD, -_roll * delta)

	var target_speed := BOOST_MULTIPLIER * FORWARD_SPEED if boosting else FORWARD_SPEED
	var acceleration := BOOST_ACCELERATION if boosting else BASE_ACCELERATION
	speed = clampf(move_toward(speed, target_speed, acceleration * delta), 0.0, MAX_SPEED)

	velocity = -global_transform.basis.z * speed
	move_and_slide()

	if global_position.y < MIN_ALTITUDE:
		global_position.y = MIN_ALTITUDE
		velocity.y = maxf(velocity.y, 0.0)
	elif global_position.y > MAX_ALTITUDE:
		global_position.y = MAX_ALTITUDE
		velocity.y = minf(velocity.y, 0.0)

	_update_visuals()


func get_speed_normalized() -> float:
	return clampf(inverse_lerp(FORWARD_SPEED, MAX_SPEED, speed), 0.0, 1.0)


func reset_transform() -> void:
	global_position = Vector3.ZERO
	rotation = Vector3.ZERO
	velocity = Vector3.ZERO
	speed = FORWARD_SPEED
	_pitch = 0.0
	_yaw = 0.0
	_roll = 0.0


func _update_visuals() -> void:
	var speed_ratio := get_speed_normalized()

	if _hull != null:
		_hull.rotation.z = (_roll / MAX_ROLL_RATE) * 0.25

	if _engine_glow != null:
		var flicker := 0.9 + randf() * 0.2
		var length := 1.0 + speed_ratio * 1.6
		_engine_glow.scale = Vector3(1.0 + speed_ratio * 0.4, 1.0 + speed_ratio * 0.4, length * flicker)

	if _engine_light != null:
		_engine_light.light_energy = 1.0 + speed_ratio * 2.0
		_engine_light.light_color = Color(0.55, 0.75, 1.0) if boosting else Color(0.45, 0.85, 1.0)
